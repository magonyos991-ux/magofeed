// AGENT 1 (image) — LA BOISSON EST CHOISIE PAR LE CALENDRIER, PAS PAR L'IA.
//
// Avant, le modele choisissait dans le menu avec la consigne « pas deux fois
// la meme ». Une consigne s'oublie : deux Pocari sont sortis la meme semaine.
// Maintenant : rotation fixe sur 12 boissons, une par jour — une repetition
// est MATHEMATIQUEMENT impossible en moins de 12 jours. L'IA ne decide plus
// que de l'angle du texte (anecdote, question, defi...).
const prev = $('Memoire et aiguillage').first().json;
const MENU = [
  { slug: "guarana-antarctica", nom: "Guaraná Antarctica", info: "soda guarana, le prefere du Bresil depuis 1921" },
  { slug: "ramune", nom: "Ramune", info: "soda japonais avec une bille de verre dans le goulot" },
  { slug: "calpico", nom: "Calpico", info: "soda lacte japonais au gout yaourt, cree en 1919" },
  { slug: "mogu-mogu", nom: "Mogu Mogu", info: "jus thailandais avec cubes de nata de coco qui se machent" },
  { slug: "inca-kola", nom: "Inca Kola", info: "soda dore peruvien qui bat le cola chez lui" },
  { slug: "milkis", nom: "Milkis", info: "soda lacte coreen doux et petillant" },
  { slug: "mountain-dew", nom: "Mountain Dew", info: "le soda vert culte des Etats-Unis, version americaine introuvable en Europe" },
  { slug: "ciao-energy", nom: "Ciao Energy", info: "energy drink introuvable dont tout le monde parle" },
  { slug: "jarritos", nom: "Jarritos", info: "sodas mexicains aux vrais fruits depuis 1950" },
  { slug: "sumol", nom: "Sumol", info: "soda portugais a l'ananas, culte a Lisbonne" },
  { slug: "pocari-sweat", nom: "Pocari Sweat", info: "boisson desalterante n°1 au Japon" },
  { slug: "vimto", nom: "Vimto", info: "boisson fruitee anglaise de Manchester, depuis 1908" },
  { slug: "chupa-chups", nom: "Chupa Chups Sparkling", info: "la sucette devenue soda petillant, Coree" }
];
// Ordre volontairement melange pour alterner les continents d'un jour a l'autre.
const ORDRE = ["ramune","jarritos","milkis","guarana-antarctica","sumol","pocari-sweat",
               "mountain-dew","mogu-mogu","vimto","calpico","ciao-energy","inca-kola","chupa-chups"];
/* Le workflow poste DEUX images par jour (13h et 19h). Une rotation par jour
   aurait donne la meme boisson aux deux posts. On avance donc d'un cran par
   CRENEAU : 13h = creneau pair, 19h = creneau impair. Jamais deux fois la
   meme boisson le meme jour, et chaque boisson revient tous les 6 jours. */
const jour = Math.floor(Date.now() / 86400000);          // numero du jour (UTC)
const creneau = new Date().getUTCHours() >= 15 ? 1 : 0;  // 19h Bruxelles = apres 15h UTC
let idx = (jour * 2 + creneau) % ORDRE.length;

/* SECOND VERROU : la memoire des posts deja publies.
   La rotation par calendrier suffit pour les passages automatiques, mais elle
   redonne la meme boisson si le workflow est relance a la main dans le meme
   creneau (un test « Execute workflow », par exemple). On regarde donc la
   liste des sujets deja publies : si la boisson du calendrier y figure, on
   avance d'un cran, autant de fois qu'il le faut. Un doublon devient
   impossible, meme en cas de relance manuelle. */
const dejaTxt = String(prev.usedTopicsList || "").toLowerCase();
const sansAccent = (t) => String(t).toLowerCase()
  .normalize("NFD").replace(/[\u0300-\u036f]/g, "");
const dejaHay = sansAccent(dejaTxt);
const dejaPubliee = (s) => {
  const item = MENU.find(x => x.slug === s);
  if (!item) return false;
  /* On compare sur le PREMIER MOT du nom (ramune, jarritos, guarana, pocari,
     mountain, chupa...), distinctif pour chacune des treize boissons. Chercher
     le nom complet echouait des que le modele ecrivait « Chupa Chups » sans
     « Sparkling » : la boisson passait pour inedite et revenait en double. */
  const cle = sansAccent(item.nom).split(/\s+/)[0];
  return cle.length >= 4 && dejaHay.includes(cle);
};
for (let tour = 0; tour < ORDRE.length && dejaPubliee(ORDRE[idx]); tour++) {
  idx = (idx + 1) % ORDRE.length;
}
const slug = ORDRE[idx];
const m = MENU.find(x => x.slug === slug);
const prompt =
  "Tu es le responsable marketing de Magofeed, appli belge (Bruxelles) qui aide a TROUVER des boissons rares et exotiques SANS ALCOOL dans les magasins proches. Public familial, ton fun et curieux, langue: FRANCAIS.\n\n" +
  "LA BOISSON DU JOUR EST IMPOSEE (ne la change sous aucun pretexte) :\n" +
  "- " + m.nom + " : " + m.info + "\n\n" +
  "Cree le post Instagram + Facebook pour CETTE boisson.\n\n" +
  "REGLES ABSOLUES:\n" +
  "- JAMAIS d'alcool, meme 0.0%. Univers 100% famille. Aucun emoji.\n" +
  "- La boisson ne change pas ; ce qui doit changer, c'est L'ANGLE. Regarde les posts deja publies ci-dessous : si cette boisson y figure, prends un angle RADICALEMENT different (anecdote historique, le-savais-tu, question communaute, defi, comparaison, souvenir d'enfance, recette d'accompagnement...).\n" +
  "- CTA vers l'appli: " + prev.appLink + "\n" +
  "- Instagram: accroche forte + 8-12 hashtags. Facebook: conversationnel, 2-4 hashtags.\n\n" +
  "POSTS DEJA PUBLIES (pour varier l'angle): " + prev.usedTopicsList;
return [{ json: { claudeBody: {
  model: "claude-opus-5", max_tokens: 4000,
  output_config: { effort: "low", format: { type: "json_schema", schema: {
    "type": "object", "additionalProperties": false,
    "required": ["topic", "image_query", "caption_instagram", "caption_facebook"],
    "properties": {
      "topic": {"type": "string", "description": "Sujet court et unique du post (inclut le nom de la boisson ET l'angle choisi)"},
      "image_query": {"type": "string", "enum": [slug], "description": "slug de la boisson du jour (impose)"},
      "caption_instagram": {"type": "string", "description": "Legende Instagram: accroche forte, corps, CTA appli, 8-12 hashtags"},
      "caption_facebook": {"type": "string", "description": "Version Facebook conversationnelle, 2-4 hashtags, CTA appli"}
    }} } },
  messages: [{ role: "user", content: prompt }]
} } }];
